    end

        return "RCTRL"
    end

    if str == "LeftAlt" then
        return "LALT"
    end

    if str == "RightAlt" then
        return "RALT"

        return "CAPS"
    end

    if str == "PageUp" then

    end

    if str == "MouseButton2" then
        return "RMB"
    end

    if str == "MouseButton3" then
        return "MMB"
    end

    if str == "MouseButton4" then
        return "MB4"

        return "MB5"
    end
